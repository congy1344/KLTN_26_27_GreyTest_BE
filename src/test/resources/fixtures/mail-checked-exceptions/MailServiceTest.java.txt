    private MailService service;

    @Test
    void testSendEmail_DeliversSuppliedEmailDetails() {
        // GreyTest trace: sendEmail | STMT-1 | BR-010 -> TP-010 -> TC-013
        String to = "recipient@example.com";
        String subject = "Account notice";
        String content = "Your account is ready.";
        boolean isMultipart = false;
        boolean isHtml = true;
        JHipsterProperties.Mail mail = new JHipsterProperties.Mail();
        mail.setFrom("noreply@example.com");
        when(jHipsterProperties.getMail()).thenReturn(mail);
        MimeMessage mimeMessage = new MimeMessage(Session.getInstance(new Properties()));
        when(javaMailSender.createMimeMessage()).thenReturn(mimeMessage);
        service.sendEmail(to, subject, content, isMultipart, isHtml);
        verify(javaMailSender).send(mimeMessage);
        assertEquals(to, mimeMessage.getAllRecipients()[0].toString());
        assertEquals(subject, mimeMessage.getSubject());
        assertEquals(content, mimeMessage.getContent().toString());
    }

        when(messageSource.getMessage(titleKey, null, Locale.ENGLISH)).thenReturn(resolvedSubject);

    @Test
    void testSendEmail_SendsConfiguredHtmlMessage() {
        // GreyTest trace: sendEmail | TRY-1::SUCCESS | BR-011 -> TP-011 -> TC-014
        String to = "recipient@example.com";
        String subject = "Account update";
        String content = "<p>Updated</p>";
        JHipsterProperties.Mail mail = new JHipsterProperties.Mail();
        mail.setFrom("sender@example.com");
        when(jHipsterProperties.getMail()).thenReturn(mail);
        MimeMessage mimeMessage = new MimeMessage(Session.getInstance(new Properties()));
        when(javaMailSender.createMimeMessage()).thenReturn(mimeMessage);
        service.sendEmail(to, subject, content, false, true);
        verify(javaMailSender).send(mimeMessage);
        assertEquals(to, mimeMessage.getAllRecipients()[0].toString());
        assertEquals("sender@example.com", mimeMessage.getFrom()[0].toString());
        assertEquals(subject, mimeMessage.getSubject());
        assertEquals(content, mimeMessage.getContent().toString());
    }

    @Test
    void testSendEmail_SwallowsMailException() {
        // GreyTest trace: sendEmail | TRY-1::CATCH-1 | BR-011 -> TP-011 -> TC-015
        JHipsterProperties.Mail mail = new JHipsterProperties.Mail();
        mail.setFrom("sender@example.com");
        when(jHipsterProperties.getMail()).thenReturn(mail);
        MimeMessage mimeMessage = new MimeMessage(Session.getInstance(new Properties()));
        when(javaMailSender.createMimeMessage()).thenReturn(mimeMessage);
        doThrow(new MailException("approved send failure") {
        }).when(javaMailSender).send(mimeMessage);
        assertDoesNotThrow(() -> service.sendEmail("recipient@example.com", "Account update", "<p>Updated</p>", false, true));
        verify(javaMailSender).send(mimeMessage);
    }

    @Test
    void testSendEmailFromTemplate_ProcessesAndSendsRequestedTemplate() {
        // GreyTest trace: sendEmailFromTemplate | STMT-1 | BR-012 -> TP-012 -> TC-016
        User user = new User();
        user.setEmail("recipient@example.com");
        user.setLangKey("en");
        String templateName = "mail/notice";
        String titleKey = "email.notice.title";
        JHipsterProperties.Mail mail = new JHipsterProperties.Mail();
        mail.setBaseUrl("https://example.com");
        mail.setFrom("sender@example.com");
        when(jHipsterProperties.getMail()).thenReturn(mail);
        when(springTemplateEngine.process(org.mockito.ArgumentMatchers.eq(templateName), any(Context.class))).thenReturn("<p>Notice</p>");
        when(messageSource.getMessage(titleKey, null, Locale.forLanguageTag("en"))).thenReturn("Notice");
        MimeMessage mimeMessage = new MimeMessage(Session.getInstance(new Properties()));
        when(javaMailSender.createMimeMessage()).thenReturn(mimeMessage);
        service.sendEmailFromTemplate(user, templateName, titleKey);
        ArgumentCaptor<Context> contextCaptor = ArgumentCaptor.forClass(Context.class);
        verify(springTemplateEngine).process(org.mockito.ArgumentMatchers.eq(templateName), contextCaptor.capture());
        assertEquals(user, contextCaptor.getValue().getVariable("user"));
        assertEquals("https://example.com", contextCaptor.getValue().getVariable("baseUrl"));
        verify(messageSource).getMessage(titleKey, null, Locale.forLanguageTag("en"));
        verify(javaMailSender).send(mimeMessage);
        assertEquals("recipient@example.com", mimeMessage.getAllRecipients()[0].toString());
        assertEquals("Notice", mimeMessage.getSubject());
        assertEquals("<p>Notice</p>", mimeMessage.getContent().toString());
    }

    @Test
    void testSendEmailFromTemplate_SkipsWhenEmailIsNull() {
        // GreyTest trace: sendEmailFromTemplate | IF-1-TRUE | BR-013 -> TP-013 -> TC-017
        User user = new User();

        user.setLangKey("en");
        service.sendEmailFromTemplate(user, "mail/notice", "email.notice.title");
        verifyNoMoreInteractions(jHipsterProperties, javaMailSender, messageSource, springTemplateEngine);
    }
